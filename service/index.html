💻 JSX - ឧទាហរណ៍លម្អិត

របៀបសរសេរ HTML ក្នុង JavaScript រៀងរាល់ថ្ងៃ

ឧទាហរណ៍ ១ - ជំងឺើង្ហា JSX សាមញ្ញ

ដើម្បីបង្ហាញឈ្មោះ និងអាយុ

import React from 'react';

function Greeting() {
  const name = "សុខា";
  const age = 25;

  return (
    <>
      <h1>សូស្វាគមន៍ {name}</h1>
      <p>អាយុ: {age} ឆ្នាំ</p>
    </>
  );
}

export default Greeting;

💡 ពន្យល់:

  • {name} បង្ហាញតម្លៃ variable "សុខា"
  • {age} បង្ហាញលេខ 25
  • <> </> ជា Fragment (ឲ្យម្ដង root element)

ឧទាហរណ៍ ២ - Conditional Rendering

បង្ហាញរឺលាក់ដោយផ្អែកលើលក្ខខណ្ឌ

function LoginStatus() {
  const isLoggedIn = true;

  return (
    <div>
      {isLoggedIn ? (
        <h1>សូស្វាគមន៍ ឡើងវិញ! 🎉</h1>
      ) : (
        <h1>សូមចូលប្រើប្រាស់ឡើង</h1>
      )}
    </div>
  );
}

💡 ដូចម្ដេច ដំណើរការ:

ប្រសិនបើ isLoggedIn ពិត → បង្ហាញ "សូស្វាគមន៍ឡើងវិញ!"
ប្រសិនបើ មិនពិត → បង្ហាញ "សូមចូលប្រើប្រាស់ឡើង"

ឧទាហរណ៍ ៣ - ប្រើប្រាស់ Array ដែលមាន .map()

បង្ហាញបញ្ជីលក្ខខ័ណ្ឌ

function StudentList() {
  const students = [
    { id: 1, name: "សុខា" },
    { id: 2, name: "សម័យ" },
    { id: 3, name: "មរតី" }
  ];

  return (
    <ul>
      {students.map((student) => (
        <li key={student.id}>
          {student.name}
        </li>
      ))}
    </ul>
  );
}

⚠️ សម្គាល់ថា:

key={student.id} ពិតជាចាំបាច់! វាដាក់សម្គាល់របស់ component និងទើបកែក្នុងក្តារ DOM ឱ្យប្រកបដោយប្រសិទ្ធភាព

ឧទាហរណ៍ ៣ - Destructuring Props (ងាយស្រួលជាង)

ឡើងវិញ ដូចប៉ុន្តែលេខយូច្បាក់ ក្នុង component

// Destructuring = ច្របាច់ដក props ចេញ
function ProductCard({ name, price, description }) {
  // គ្មាន props.name ទេ - ដែលយូច្បាក់ដក name ចេញរួច
  return (
    <div>
      <h2>{name}</h2>
      <p>តម្លៃ: ${price}</p>
      <p>{description}</p>
    </div>
  );
}

⚠️ ចាប់ផ្តើម:

Destructuring ជ្រើសយក props ដែលត្រូវការ។ ងាយស្រួលនិងលេខយូច្បាក់នាំ}

⚡ useState - គ្រប់គ្រងទិន្នន័យ

အခြား State ដែលផ្លាស់ប្ដូរដូចជា Counter, Form

ឧទាហរណ៍ ១ - Counter (រាប់ម្ដង)

import { useState } from 'react';

function Counter() {
  // ═════════════════════════════════════════
  // useState = Hook គ្រប់គ្រង state
  // ═════════════════════════════════════════
  const [count, setCount] = useState(0);
  // count = តម្លៃបច្ចុប្បន្ន (0)
  // setCount = function ឯើងផ្លាស់ប្ដូរ count

  return (
    <div>
      <h1>រាប់រវាង: {count}</h1>

      <button onClick={() => setCount(count + 1)}>
        ម្ដង + 1
      </button>

      <button onClick={() => setCount(count - 1)}>
        ម្ដង - 1
      </button>
    </div>
  );
}

🎯 វា​ដំណើរការ​ដូច​ម្ដេច:

  1. ចុច button "ម្ដង + 1" → count ក្លាយជា 1
  2. React render ឡើងវិញ ហើយបង្ហាញ 1
  3. ចុច button "ម្ដង - 1" → count ក្លាយជា 0
  4. React render ឡើងវិញ ហើយបង្ហាញ 0

ឧទាហរណ៍ ២ - Form (Controlled Input)

គ្រប់គ្រង input value ដោយ React

function LoginForm() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const handleSubmit = (e) => {
    e.preventDefault();
    console.log("Email:", email, "Password:", password);
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        placeholder="ឈ្មោះប្រើប្រាស់"
      />

      <input
        type="password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
        placeholder="ពាក្យសម្ងាត់"
      />

      <button type="submit">ចូលប្រើប្រាស់</button>
    </form>
  );
}

💡 ចំនួច្បាក់៖

  • onChange ពេល input ផ្លាស់ប្ដូរ → ឆ្លងទៅ setEmail/setPassword
  • e.preventDefault() កុំឱ្យ submit រលូត ទំព័រ
  • value={email} ធានាថា input រក្សា state React

🪝 useEffect - គេហៅឡើង Side Effects

រត់ក្នុងនាង fetch data, timer, localStorage

ឧទាហរណ៍ ១ - ឡើង ម្ដង Render

ដូចជា useEffect រត់ម្ដង ពេលនឹងបង្ហាញ component

import { useState, useEffect } from 'react';

function PageTitle() {
  const [title, setTitle] = useState("ស្ដីឯង");

  useEffect(() => {
    console.log("Component បង្ហាញ!");
    document.title = title;
  }, []); // [] = រត់ម្ដង

  return <h1>{title}</h1>;
}

ឧទាហរណ៍ ២ - Fetch ទិន្នន័យពីមាន API

function GetUsers() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Fetch ទិន្នន័យពីមាន server
    fetch("https://jsonplaceholder.typicode.com/users")
      .then(res => res.json())
      .then(data => {
        setUsers(data);
        setLoading(false);
      });
  }, []); // រត់ម្ដង ពេលមាន

  if (loading) return <p>រង់ចាំ...</p>;

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}

💡 អ្វីដែលផ្តើម៖

  • useEffect រត់បន្ទាប់ពី component render
  • fetch ទាញទិន្នន័យ រៀងរាល់ 2-3 វិនាទី
  • setUsers រក្សា ទិន្នន័យក្នុង state
  • [] dependency array មានន័យ "រត់ម្ដង"

ឧទាហរណ៍ ៣ - Timer/Interval

function Clock() {
  const [time, setTime] = useState(0);

  useEffect(() => {
    // រត់រាល់ 1 វិនាទី
    const interval = setInterval(() => {
      setTime(t => t + 1);
    }, 1000);

    // 청소ន = បន្ដឹងលុប interval ពេលមិនត្រូវការ
    return () => clearInterval(interval);
  }, []);

  return <h1>ពេលវេលា: {time}</h1>;
}

📋 Dependency Array អត្ថន័យ

useEffect(() => {}, [])

ឡើង ម្ដង ពេល component mount (ដូច componentDidMount)

useEffect(() => {}, [count])

រត់រាល់ពេល count ផ្លាស់ប្ដូរ

useEffect(() => {})

រត់រាល់ពេល render (⚠️ ប្រុងប្រយ័ត្ន - អាចធ្វើ infinite loop!)

🛒 Real Project - E-Commerce Store

គម្រោង ពិតប្រាកដ - ល្ក្ខ ការគាំង គៀង Cart

គម្រោងពេញលេញ - មាន Items, Quantity, Total

import { useState } from 'react';

function EcommerceApp() {
  // ═════════════════════════════════════════
  // ១. ទិន្នន័យផលិតផល
  // ═════════════════════════════════════════
  const products = [
    { id: 1, name: "Nike Shoes", price: 120 },
    { id: 2, name: "iPhone 14", price: 999 },
    { id: 3, name: "Laptop", price: 1200 }
  ];

  // ═════════════════════════════════════════
  // ២. Cart State
  // ═════════════════════════════════════════
  const [cart, setCart] = useState([]);

  // ═════════════════════════════════════════
  // ៣. បង្កើត functions
  // ═════════════════════════════════════════
  const addToCart = (product) => {
    const existingItem = cart.find(item => item.id === product.id);

    if (existingItem) {
      // ប្រសិនបើមាននៅក្នុង cart រួច - បង្កើន quantity
      setCart(cart.map(item =>
        item.id === product.id
          ? { ...item, quantity: item.quantity + 1 }
          : item
      ));
    } else {
      // ផលិតផលថ្មី - បន្ថែមទៅ cart
      setCart([...cart, { ...product, quantity: 1 }]);
    }
  };

  const removeFromCart = (productId) => {
    setCart(cart.filter(item => item.id !== productId));
  };

  const getTotalPrice = () => {
    return cart.reduce((total, item) =>
      total + item.price * item.quantity,
      0
    );
  };

  return (
    <div>
      <h1>ហាង E-Commerce 🛒</h1>

      <h2>ផលិតផល</h2>
      {products.map((product) => (
        <div key={product.id} style={{ border: '1px solid #ccc', padding: '10px' }}>
          <h3>{product.name}</h3>
          <p>${product.price}</p>
          <button onClick={() => addToCart(product)}>
            បន្ថែមទៅក្រាប់
          </button>
        </div>
      ))}

      <h2>ក្រាប់របស់ខ្ញុំ</h2>
      {cart.length === 0 ? (
        <p>ក្រាប់ទទេ</p>
      ) : (
        <>
          {cart.map((item) => (
            <div key={item.id}>
              <p>{item.name} x{item.quantity} = ${item.price * item.quantity}</p>
              <button onClick={() => removeFromCart(item.id)}>
                លុប
              </button>
            </div>
          ))}
          <h3>សរុប: ${getTotalPrice()}</h3>
        </>
      )}
    </div>
  );
}

export default EcommerceApp;

🎯 របៀបដំណើរការ៖

  1. ចុច "បន្ថែមទៅក្រាប់" ដើម្បីបង្កើនផលិតផល
  2. ប្រសិនបើផលិតផលដូចគ្នា - quantity បង្កើន
  3. គណនា total price ដោយប្រើ reduce()
  4. ចុច "លុប" ដើម្បីដកផលិតផលចេញ

🎓 គោលការណ៍សក្ខី ដែលប្រើ:

  • useState - ដាក់ cart ក្នុង state
  • .map() - បង្ហាញបញ្ជីផលិតផល និង cart items
  • Conditional Rendering - បង្ហាញ "ក្រាប់ទទេ" ឬ items
  • .find() - ស្វាគមន៍ផលិតផល នៅក្នុង cart
  • .filter() - លុបផលិតផល
  • .reduce() - គណនា total price
  • Spread Operator (...) - ច្របាច់ array / object
