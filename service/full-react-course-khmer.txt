5. useReducer

ប្រើ state logic ស្មុគស្មាញ

Syntax
const [state, dispatch] = useReducer(reducer, initialState);
Example
import { useReducer } from "react";

function reducer(state, action) {
  switch(action.type) {
    case "increment":
      return { count: state.count + 1 };

    case "decrement":
      return { count: state.count - 1 };

    default:
      return state;
  }
}

function App() {
  const [state, dispatch] = useReducer(reducer, { count: 0 });

  return (
    <div>
      <h1>{state.count}</h1>

      <button onClick={() => dispatch({ type: "increment" })}>
        +
      </button>
    </div>
  );
}
When useReducer?

✅ Complex state
✅ Multiple actions
✅ Shopping cart
✅ Big forms

useState vs useReducer

Small state	Big logic
Easy	Organized

# Fetch API Example

```jsx

  const [users, setUsers] = useState([]);

      ))}
    </div>
  );
}
```

# 5. useReducer

Complex State Logic

Syntax:

```jsx
const [state, dispatch] = useReducer(reducer, initialState);
```

---

# Example

```jsx
import { useReducer } from "react";

function reducer(state, action) {
  switch(action.type) {
    case "increment":
      return {
        count: state.count + 1
      };

    case "decrement":
      return {
        count: state.count - 1
      };

    default:
      return state;
  }
}

function App() {
  const [state, dispatch] = useReducer(reducer, {
    count: 0
  });

  return (
    <div>
      <h1>{state.count}</h1>

      <button onClick={() => dispatch({ type: "increment" })}>
        +
      </button>
    </div>
  );
}
```

---

# useReducer When to Use?

✅ Complex State
✅ Shopping Cart
✅ Big Forms
✅ Multiple Actions

---

# useState vs useReducer

| useState    | useReducer |
| ----------- | ---------- |
| Simple      | Complex    |
| Easy        | Organized  |
| Small State | Big Logic  |
